class Employee(models.Model):
    employee_id = models.CharField(
        "Número de empleado",
        max_length=20,
        unique=True,
    )
    first_name = models.CharField("Nombre(s)", max_length=100)
    last_name = models.CharField("Apellidos", max_length=100)
    is_active = models.BooleanField("Activo", default=True)

    class Meta:
        verbose_name = "Trabajador"
        verbose_name_plural = "Trabajadores"
        ordering = ["last_name", "first_name"]

    def __str__(self):
        return f"{self.employee_id} - {self.first_name} {self.last_name}"

        editable=False,
    )
    issued_at = models.DateTimeField(
        "Fecha de entrega",
        default=timezone.now,
    )
    requires_return = models.BooleanField(
        "Requiere devolución",
        default=True,
        editable=False,
    )
    notes = models.TextField(
        "Observaciones",
        blank=True,
    )

    class Meta:
        verbose_name = "Entrega"
        verbose_name_plural = "Entregas"
        ordering = ["-issued_at", "-pk"]
        constraints = [
            models.CheckConstraint(
                condition=models.Q(quantity__gte=1),
                name="quantity_issue_quantity_gte_1",
            ),
            models.CheckConstraint(
                condition=models.Q(
                    returned_quantity__lte=models.F("quantity")
                ),
                name="quantity_issue_returned_lte_quantity",
            ),
            models.CheckConstraint(
                condition=(
                    models.Q(stock__isnull=False, tool__isnull=True)
                    | models.Q(stock__isnull=True, tool__isnull=False)
                ),
                name="issue_stock_or_tool",
            ),
            models.CheckConstraint(
                condition=(
                    models.Q(tool__isnull=True)
                    | models.Q(quantity=1)
                ),
                name="issue_individual_quantity_one",
            ),
            models.UniqueConstraint(
                fields=["tool"],
                condition=models.Q(
                    tool__isnull=False,
                    requires_return=True,
                    returned_quantity=0,
                ),
                name="unique_open_issue_per_tool",
                violation_error_message=(
                    "Esta pieza ya tiene una entrega pendiente."
                ),
            ),
        ]

    
    @property
    def fecha_limite(self):
        from .loan_rules import fecha_limite_prestamo
        return fecha_limite_prestamo(self)

    @property
    def estado_plazo(self):
        from .loan_rules import estado_plazo_prestamo
        return estado_plazo_prestamo(self)

    def __str__(self):
        if self.tool_id:
            reference = self.tool.tool_code
        elif self.stock_id:
            reference = self.stock.article.article_code
        else:
            reference = "Sin artículo"

        return (
            f"Entrega #{self.pk} — {reference} — "
            f"Trabajador {self.employee.employee_id}"
        )

class QuantityReturn(models.Model):
    class Condition(models.TextChoices):
        GOOD = "good", "Buen estado"
        DAMAGED = "damaged", "Dañada"

    issue = models.ForeignKey(
        QuantityIssue,
        on_delete=models.PROTECT,
        verbose_name="Entrega original",
        related_name="returns",
    )
    quantity = models.PositiveIntegerField(
        "Cantidad devuelta",
        validators=[MinValueValidator(1)],
    )
    resolved_quantity = models.PositiveIntegerField(
        "Cantidad resuelta acumulada",
        default=0,
        editable=False,
    )
    condition = models.CharField(
        "Condición al regresar",
        max_length=20,
        choices=Condition.choices,
    )
    returned_at = models.DateTimeField(
        "Fecha de devolución",
        default=timezone.now,
    )
    reason = models.CharField(
        "Motivo del daño",
        max_length=200,
        blank=True,
    )
    notes = models.TextField(
        "Observaciones",
        blank=True,
    )

    class Meta:
        verbose_name = "Devolución"
        verbose_name_plural = "Devoluciones"
        ordering = ["-returned_at", "-pk"]
        constraints = [
            models.CheckConstraint(
                condition=models.Q(quantity__gte=1),
                name="quantity_return_quantity_gte_1",
            ),
            models.CheckConstraint(
                condition=models.Q(
                    resolved_quantity__lte=models.F("quantity")
                ),
                name="quantity_return_resolved_lte_quantity",
            ),
        ]

    def clean(self):
        super().clean()
        if (
            self.condition == self.Condition.DAMAGED
            and not self.reason.strip()
        ):
            raise ValidationError({
                "reason": "Indica el motivo del daño.",
            })

    def __str__(self):
        return (
            f"Entrega #{self.issue_id} — "
            f"Devolución de {self.quantity} piezas"
        )

"""Reglas centrales de préstamos: vencimiento y duplicidad por artículo.

Esta versión no efectúa movimientos de inventario ni consume autorizaciones.
"""
from datetime import timedelta

from django.core.exceptions import ValidationError
from django.db.models import F, Q
from django.utils import timezone

from .models import Article, Loan, MaterialRequest, MaterialRequestLine, QuantityIssue

PLAZO_PRESTAMO_DIAS = 15
ESTADOS_ACTIVOS = (
    MaterialRequest.Status.SUBMITTED,
    MaterialRequest.Status.PREPARED,
)


def entregas_vencidas(employee_id, *, now=None):
    """Entregas retornables con saldo pendiente tras 15 días naturales."""
    ahora = now or timezone.now()
    return QuantityIssue.objects.filter(
        employee_id=employee_id,
        requires_return=True,
        returned_quantity__lt=F("quantity"),
        issued_at__lt=ahora - timedelta(days=PLAZO_PRESTAMO_DIAS),
    )


def prestamos_antiguos_vencidos(employee_id, *, now=None):
    """Préstamos legacy aún no unificados, sin duplicar QuantityIssue."""
    ahora = now or timezone.now()
    return Loan.objects.filter(
        employee_id=employee_id,
        returned_at__isnull=True,
        unified_issue__isnull=True,
        borrowed_at__lt=ahora - timedelta(days=PLAZO_PRESTAMO_DIAS),
    )


def tiene_adeudos_vencidos(employee_id, *, now=None):
    return (
        entregas_vencidas(employee_id, now=now).exists()
        or prestamos_antiguos_vencidos(employee_id, now=now).exists()
    )


def validar_bloqueo_vencimiento(employee_id, *, now=None):
    """Bloqueo general para NUEVAS comandas si existen adeudos vencidos."""
    if tiene_adeudos_vencidos(employee_id, now=now):
        raise ValidationError(
            "Tienes préstamos vencidos (más de 15 días). "
            "Devuelve los artículos pendientes para solicitar nuevamente."
        )


def motivo_duplicado(*, employee_id, article_id, excluir_solicitud_id=None):
    """Retorna el motivo de duplicidad, o None si no existe conflicto."""
    articulo = Article.objects.filter(pk=article_id, is_active=True).first()
    if articulo is None:
        raise ValidationError("El artículo no existe o está inactivo.")

    solicitudes = MaterialRequestLine.objects.filter(
        request__employee_id=employee_id,
        request__status__in=ESTADOS_ACTIVOS,
        article_id=article_id,
    )
    if excluir_solicitud_id is not None:
        solicitudes = solicitudes.exclude(request_id=excluir_solicitud_id)
    if solicitudes.exists():
        return "Ya existe una solicitud activa de este artículo."

    # Los consumibles se restringen durante una solicitud activa, pero
    # no generan deuda de devolución tras ser entregados.
    if articulo.return_mode != Article.ReturnMode.RETURNABLE:
        return None

    if QuantityIssue.objects.filter(
        employee_id=employee_id,
        requires_return=True,
        returned_quantity__lt=F("quantity"),
    ).filter(
        Q(stock__article_id=article_id) | Q(tool__article_id=article_id)
    ).exists():
        return "Ya tienes este artículo pendiente de devolución."

    if Loan.objects.filter(
        employee_id=employee_id,
        returned_at__isnull=True,
        unified_issue__isnull=True,
        tool__article_id=article_id,
    ).exists():
        return "Ya tienes un préstamo anterior de este artículo pendiente."

    return None


def validar_sin_duplicado(*, employee_id, article_id, excluir_solicitud_id=None):
    motivo = motivo_duplicado(
        employee_id=employee_id,
        article_id=article_id,
        excluir_solicitud_id=excluir_solicitud_id,
    )
    if motivo:
        raise ValidationError(
            f"{motivo} Se requiere autorización del jefe para el duplicado."
        )


def fecha_limite(issue):
    if not issue.requires_return:
        return None
    return issue.issued_at + timedelta(days=PLAZO_PRESTAMO_DIAS)


def estado_prestamo(issue, *, now=None):
    if not issue.requires_return:
        return "consumible"
    if issue.returned_quantity >= issue.quantity:
        return "devuelto"
    restante = fecha_limite(issue) - (now or timezone.now())
    if restante.total_seconds() < 0:
        return "vencido"
    if restante <= timedelta(days=3):
        return "por_vencer"
    return "en_plazo"


def validar_articulo_solicitable(
    *,
    employee_id,
    article_id,
    exclude_request_id=None,
):
    """
    Compatibilidad con las funciones que todavía
    utilizan el nombre anterior.
    """

    validar_bloqueo_vencimiento(employee_id)

    validar_sin_duplicado(
        employee_id=employee_id,
        article_id=article_id,
        excluir_solicitud_id=exclude_request_id,
    )

from django.core.exceptions import ValidationError
from django.db import transaction
from django.utils import timezone

from .loan_rules import (
    validar_bloqueo_vencimiento,
    validar_sin_duplicado,
)
from .models import Employee, MaterialRequest, MaterialRequestLine
from .selectors import catalogo_articulos


def bloquear_trabajador(employee_id):
    """Comprueba que el trabajador exista y esté activo."""
    employee = Employee.objects.select_for_update().filter(
        pk=employee_id,
        is_active=True,
    ).first()

    if employee is None:
        raise ValidationError("El trabajador no está activo.")

    return employee


def set_cart_quantity(*, employee_id, article_id, quantity):
    if type(quantity) is not int or quantity < 1:
        raise ValidationError(
            "La cantidad debe ser un número entero mayor que cero."
        )

    with transaction.atomic():
        bloquear_trabajador(employee_id)

        # No permitir nuevas solicitudes con adeudos vencidos.
        validar_bloqueo_vencimiento(employee_id)

        cart, _ = MaterialRequest.objects.get_or_create(
            employee_id=employee_id,
            status=MaterialRequest.Status.DRAFT,
        )

        updated = MaterialRequest.objects.filter(
            pk=cart.pk,
            employee_id=employee_id,
            status=MaterialRequest.Status.DRAFT,
        ).update(updated_at=timezone.now())

        if updated != 1:
            raise ValidationError(
                "Este carrito ya no está abierto. Recarga la página."
            )

        article = catalogo_articulos().filter(
            pk=article_id,
        ).first()

        if article is None:
            raise ValidationError(
                "El artículo no existe o está inactivo."
            )

        # No permitir duplicados sin autorización.
        validar_sin_duplicado(
            employee_id=employee_id,
            article_id=article_id,
            excluir_solicitud_id=cart.pk,
        )

        if quantity > article.available_quantity:
            raise ValidationError(
                f"Actualmente hay {article.available_quantity} "
                "piezas disponibles de este artículo."
            )

        line, _ = MaterialRequestLine.objects.update_or_create(
            request=cart,
            article=article,
            defaults={"requested_quantity": quantity},
        )

        return line


def submit_cart(*, employee_id, cart_id):
    with transaction.atomic():
        bloquear_trabajador(employee_id)

        validar_bloqueo_vencimiento(employee_id)

        cart = MaterialRequest.objects.select_for_update().filter(
            pk=cart_id,
            employee_id=employee_id,
            status=MaterialRequest.Status.DRAFT,
        ).first()

        if cart is None:
            raise ValidationError(
                "El carrito ya fue enviado o no está disponible."
            )

        lines = MaterialRequestLine.objects.filter(
            request_id=cart.pk,
        )

        if not lines.exists():
            raise ValidationError(
                "Agrega al menos un artículo antes de enviar la solicitud."
            )

        for article_id in lines.values_list(
            "article_id", flat=True
        ):
            validar_sin_duplicado(
                employee_id=employee_id,
                article_id=article_id,
                excluir_solicitud_id=cart.pk,
            )

        if lines.filter(article__is_active=False).exists():
            raise ValidationError(
                "Tu carrito contiene un artículo desactivado. "
                "Retíralo antes de enviar la solicitud."
            )

        if lines.filter(prepared_quantity__gt=0).exists():
            raise ValidationError(
                "El carrito tiene cantidades preparadas inesperadas. "
                "Solicita al almacenista que lo revise."
            )

        submitted_at = timezone.now()

        updated = MaterialRequest.objects.filter(
            pk=cart.pk,
            employee_id=employee_id,
            status=MaterialRequest.Status.DRAFT,
        ).update(
            status=MaterialRequest.Status.SUBMITTED,
            submitted_at=submitted_at,
            updated_at=submitted_at,
        )

        if updated != 1:
            raise ValidationError(
                "El carrito ya fue enviado o no está disponible."
            )

        return MaterialRequest.objects.get(pk=cart.pk)

from django.core.exceptions import PermissionDenied, ValidationError
from django.db import transaction
from django.utils import timezone
from django.db.models import F
from .loan_rules import validar_articulo_solicitable

from .models import (
    Article,
    MaterialRequest,
    MaterialRequestLine,
    MaterialRequestPiece,
    EmployeeCredential,
    QuantityIssue,
    QuantityStock,
)
from .selectors import catalogo_articulos, piezas_individuales_disponibles
from .credentials import verify_employee_pin
from .services import register_quantity_issue

def prepare_request_line(
    *,
    line_id,
    quantity,
    operator,
    tool_ids=None,
):
    if not (
        operator.is_active
        and operator.is_staff
        and operator.has_perm("stockroom.change_materialrequest")
    ):
        raise PermissionDenied(
            "No tienes permiso para preparar solicitudes."
        )

    if type(quantity) is not int or quantity < 0:
        raise ValidationError(
            "La cantidad preparada debe ser un entero igual o mayor que cero."
        )

    selected_ids = list(tool_ids or [])

    if any(type(tool_id) is not int for tool_id in selected_ids):
        raise ValidationError("La selección de piezas no es válida.")

    if len(selected_ids) != len(set(selected_ids)):
        raise ValidationError("No puedes seleccionar la misma pieza dos veces.")

    with transaction.atomic():
        updated = MaterialRequest.objects.filter(
            lines__pk=line_id,
            status=MaterialRequest.Status.SUBMITTED,
            employee__is_active=True,
        ).update(updated_at=timezone.now())

        if updated != 1:
            raise ValidationError(
                "La solicitud no está disponible para preparación."
            )

        line = MaterialRequestLine.objects.select_related(
            "article"
        ).get(pk=line_id)

        if quantity > line.requested_quantity:
            raise ValidationError(
                "No puedes preparar más piezas de las solicitadas."
            )

        if quantity == 0:
            if selected_ids:
                raise ValidationError(
                    "Si no entregarás piezas, deja vacía la selección."
                )
        else:
            article = catalogo_articulos().filter(
                pk=line.article_id,
            ).first()

            if article is None:
                raise ValidationError("El artículo está inactivo.")

            if quantity > article.available_quantity:
                raise ValidationError(
                    f"Actualmente hay {article.available_quantity} "
                    "piezas disponibles."
                )

            if article.tracking_mode == Article.TrackingMode.INDIVIDUAL:
                if len(selected_ids) != quantity:
                    raise ValidationError(
                        "Selecciona una pieza individual por cada unidad preparada."
                    )

                available_count = piezas_individuales_disponibles().filter(
                    article_id=article.pk,
                    pk__in=selected_ids,
                ).count()

                if available_count != quantity:
                    raise ValidationError(
                        "Alguna pieza no está disponible o pertenece a otro artículo."
                    )
            elif selected_ids:
                raise ValidationError(
                    "Los artículos por cantidad no requieren seleccionar piezas."
                )

        line.selected_pieces.all().delete()

        MaterialRequestPiece.objects.bulk_create([
            MaterialRequestPiece(line=line, tool_id=tool_id)
            for tool_id in selected_ids
        ])

        line.prepared_quantity = quantity
        line.save(update_fields=["prepared_quantity"])

    return line

def finish_request_preparation(*, request_id, operator):
    if not (
        operator.is_active
        and operator.is_staff
        and operator.has_perm("stockroom.change_materialrequest")
    ):
        raise PermissionDenied(
            "No tienes permiso para preparar solicitudes."
        )

    with transaction.atomic():
        now = timezone.now()

        updated = MaterialRequest.objects.filter(
            pk=request_id,
            status=MaterialRequest.Status.SUBMITTED,
            employee__is_active=True,
        ).update(
            status=MaterialRequest.Status.PREPARED,
            prepared_by=operator,
            prepared_at=now,
            preparation_version=F("preparation_version") + 1,
            updated_at=now,
        )

        if updated != 1:
            raise ValidationError(
                "La solicitud ya fue preparada o no está disponible."
            )

        lines = list(
            MaterialRequestLine.objects.filter(
                request_id=request_id,
            ).select_related("article").prefetch_related(
                "selected_pieces"
            )
        )

        if not any(line.prepared_quantity > 0 for line in lines):
            raise ValidationError(
                "Prepara al menos una pieza antes de continuar."
            )

        for line in lines:
            quantity = line.prepared_quantity
            selections = list(line.selected_pieces.all())

            if quantity > line.requested_quantity:
                raise ValidationError(
                    "Hay una cantidad preparada mayor que la solicitada."
                )

            if quantity == 0:
                if selections:
                    raise ValidationError(
                        "Un artículo sin cantidad preparada tiene piezas seleccionadas."
                    )
                continue

            article = catalogo_articulos().filter(
                pk=line.article_id,
            ).first()

            if article is None:
                raise ValidationError(
                    f"El artículo {line.article.article_code} está inactivo."
                )

            if quantity > article.available_quantity:
                raise ValidationError(
                    f"No hay suficientes piezas disponibles de "
                    f"{article.article_code}. Revisa la preparación."
                )

            if article.tracking_mode == Article.TrackingMode.INDIVIDUAL:
                tool_ids = [selection.tool_id for selection in selections]

                if len(tool_ids) != quantity:
                    raise ValidationError(
                        f"Revisa las piezas seleccionadas de {article.article_code}."
                    )

                available_count = piezas_individuales_disponibles().filter(
                    article_id=article.pk,
                    pk__in=tool_ids,
                ).count()

                if available_count != quantity:
                    raise ValidationError(
                        f"Alguna pieza de {article.article_code} "
                        "ya no está disponible."
                    )
            elif selections:
                raise ValidationError(
                    "Un artículo por cantidad tiene piezas individuales seleccionadas."
                )

        return MaterialRequest.objects.get(pk=request_id)

def confirm_request_receipt(
    *,
    request_id,
    employee_id,
    credential_id,
    pin,
    expected_version,
):
    credential = EmployeeCredential.objects.filter(
        pk=credential_id,
        employee_id=employee_id,
        is_active=True,
        employee__is_active=True,
    ).first()

    if credential is None:
        raise ValidationError("La credencial no está disponible.")

    if not MaterialRequest.objects.filter(
        pk=request_id,
        employee_id=employee_id,
        status=MaterialRequest.Status.PREPARED,
        preparation_version=expected_version,
    ).exists():
        raise ValidationError(
            "La solicitud cambió o ya no está pendiente. "
            "Vuelve al panel y revisa las cantidades actuales."
        )

    # Fuera de la transacción de inventario:
    # los intentos fallidos de NIP deben permanecer registrados.
    employee = verify_employee_pin(
        barcode=credential.barcode,
        pin=pin,
    )

    with transaction.atomic():
        now = timezone.now()

        updated = MaterialRequest.objects.filter(
            pk=request_id,
            employee_id=employee.pk,
            employee__is_active=True,
            status=MaterialRequest.Status.PREPARED,
            preparation_version=expected_version,
        ).update(
            status=MaterialRequest.Status.CLOSED,
            closed_at=now,
            updated_at=now,
        )

        if updated != 1:
            raise ValidationError(
                "La solicitud cambió o ya no está pendiente. "
                "Vuelve al panel y revisa las cantidades actuales."
            )

        if QuantityIssue.objects.filter(
            request_line__request_id=request_id,
        ).exists():
            raise ValidationError(
                "Esta solicitud ya tiene entregas registradas. "
                "Solicita al almacenista revisar el historial."
            )

        lines = list(
            MaterialRequestLine.objects.filter(
                request_id=request_id,
            ).select_related("article").prefetch_related(
                "selected_pieces__tool"
            )
        )
        # Ultima validacion antes de registrar
        # las entregas y descontar inventario.
        for line in lines:
            if line.prepared_quantity > 0:
                validar_articulo_solicitable(
                    employee_id=employee.pk,
                    article_id=line.article_id,
                    exclude_request_id=request_id,
                )

        if not any(line.prepared_quantity > 0 for line in lines):
            raise ValidationError("La solicitud no tiene piezas preparadas.")

        for line in lines:
            quantity = line.prepared_quantity
            selections = list(line.selected_pieces.all())

            if quantity > line.requested_quantity:
                raise ValidationError(
                    "Hay una cantidad preparada mayor que la solicitada."
                )

            if quantity == 0:
                if selections:
                    raise ValidationError(
                        "Hay piezas seleccionadas en un artículo no surtido."
                    )
                continue

            article = line.article

            if article.tracking_mode == Article.TrackingMode.INDIVIDUAL:
                if len(selections) != quantity:
                    raise ValidationError(
                        f"Revisa las piezas preparadas de {article.article_code}."
                    )

                for selection in selections:
                    if selection.tool.article_id != article.pk:
                        raise ValidationError(
                            "Una pieza preparada pertenece a otro artículo."
                        )

                    register_quantity_issue(
                        QuantityIssue(
                            employee_id=employee.pk,
                            tool_id=selection.tool_id,
                            quantity=1,
                            issued_at=now,
                            request_line=line,
                        )
                    )
            else:
                if selections:
                    raise ValidationError(
                        "Un artículo por cantidad tiene piezas individuales."
                    )

                stock = QuantityStock.objects.filter(
                    article_id=article.pk,
                ).first()

                if stock is None:
                    raise ValidationError(
                        f"{article.article_code} no tiene registro de existencias."
                    )

                register_quantity_issue(
                    QuantityIssue(
                        employee_id=employee.pk,
                        stock=stock,
                        quantity=quantity,
                        issued_at=now,
                        request_line=line,
                    )
                )

        return MaterialRequest.objects.get(pk=request_id)

def reopen_request_preparation(*, request_id, operator):
    if not (
        operator.is_active
        and operator.is_staff
        and operator.has_perm("stockroom.change_materialrequest")
    ):
        raise PermissionDenied(
            "No tienes permiso para corregir preparaciones."
        )

    with transaction.atomic():
        updated = MaterialRequest.objects.filter(
            pk=request_id,
            status=MaterialRequest.Status.PREPARED,
            closed_at__isnull=True,
        ).update(
            status=MaterialRequest.Status.SUBMITTED,
            preparation_version=F("preparation_version") + 1,
            prepared_by=None,
            prepared_at=None,
            updated_at=timezone.now(),
        )

        if updated != 1:
            raise ValidationError(
                "Solo puedes reabrir una preparación pendiente "
                "de confirmar recibido."
            )

        if QuantityIssue.objects.filter(
            request_line__request_id=request_id,
        ).exists():
            raise ValidationError(
                "Esta solicitud ya tiene entregas registradas. "
                "No se puede reabrir."
            )

        return MaterialRequest.objects.get(pk=request_id)
